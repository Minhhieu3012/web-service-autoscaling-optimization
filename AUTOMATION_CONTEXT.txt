import asyncio
import math
import os
import time
from fastapi import FastAPI, HTTPException, Query, Request
from prometheus_client import Counter, Gauge, Histogram
from prometheus_fastapi_instrumentator import Instrumentator

# KHỞI TẠO ỨNG DỤNG & CẤU HÌNH CON ĐO (METRICS)
app = FastAPI(
    title="Autoscaling Web Service",
    description="Dịch vụ web stateless phục vụ thực nghiệm tự động co giãn (HPA & KEDA)",
    version="1.0.0",
)

# Con đo 1: Đếm số yêu cầu đang được xử lý cùng lúc
IN_FLIGHT_REQUESTS = Gauge(
    "app_in_flight_requests",
    "Số lượng yêu cầu đang được xử lý đồng thời tại thời điểm hiện tại",
)

# Con đo 2: Đếm số yêu cầu đang phải xếp hàng chờ
QUEUE_LENGTH = Gauge(
    "app_queue_length",
    "Số lượng công việc đang nằm chờ trong hàng đợi trước khi được xử lý",
)

# Con đo 3: Đếm tổng số công việc đã hoàn thành hoặc bị lỗi
PROCESSED_JOBS_TOTAL = Counter(
    "app_processed_jobs_total",
    "Tổng số yêu cầu đã đi qua hệ thống",
    ["endpoint", "status"],
)

# Con đo 4: Đo phân phối thời gian chờ trong hàng đợi
QUEUE_WAIT_DURATION = Histogram(
    "app_queue_wait_duration_seconds",
    "Thời gian một yêu cầu phải nằm chờ trong hàng đợi (tính bằng giây)",
    buckets=[0.01, 0.05, 0.1, 0.25, 0.5, 1.0, 2.0, 5.0, 10.0],
)

# Tự động gắn bộ thu thập chuẩn (độ trễ p95, throughput, error rate)
Instrumentator().instrument(app).expose(app, endpoint="/metrics")

# Đọc số lượng worker đồng thời từ biến môi trường
# Giúp dễ dàng thay đổi tham số thực nghiệm từ file deployment.yaml của Kubernetes
MAX_CONCURRENT_WORKERS = int(os.getenv("MAX_CONCURRENT_WORKERS", "5"))
worker_semaphore = asyncio.Semaphore(MAX_CONCURRENT_WORKERS)


# BỘ ĐẾM TỰ ĐỘNG CHO MỌI YÊU CẦU ĐI VÀO/ĐI RA
@app.middleware("http")
async def track_in_flight_requests(request: Request, call_next):
    # Bỏ qua không đếm các yêu cầu kiểm tra sức khỏe hoặc cào số liệu của hệ thống
    if request.url.path in ["/metrics", "/health"]:
        return await call_next(request)

    IN_FLIGHT_REQUESTS.inc()
    try:
        response = await call_next(request)
        return response
    finally:
        IN_FLIGHT_REQUESTS.dec()

# CÁC ĐƯỜNG DẪN API (ENDPOINTS) PHỤC VỤ THỰC NGHIỆM
@app.get("/health", tags=["System"])
async def health_check():
    """Kiểm tra ứng dụng còn sống hay không (phục vụ Kubernetes Liveness/Readiness Probe)."""
    return {"status": "healthy"}

@app.get("/api/cpu", tags=["Workloads"])
def simulate_cpu_load(
    iterations: int = Query(
        default=500_000,
        ge=1_000,
        le=20_000_000,
        description="Số vòng lặp tính toán để ép xung CPU",
    )
):
    """
    Endpoint 1: Tạo tải nặng về CPU.
    Dùng để kiểm thử chính sách co giãn theo CPU (Kubernetes HPA).
    """
    start_time = time.perf_counter()

    # Thực hiện phép tính căn bậc hai và lượng giác liên tục để tiêu tốn CPU thật
    accumulator = 0.0
    for i in range(1, iterations + 1):
        accumulator += math.sqrt(i) * math.sin(i)

    duration = time.perf_counter() - start_time
    PROCESSED_JOBS_TOTAL.labels(endpoint="cpu", status="success").inc()

    return {
        "endpoint": "cpu",
        "iterations": iterations,
        "processing_time_seconds": round(duration, 4),
    }

@app.get("/api/queue", tags=["Workloads"])
async def simulate_queue_load(
    work_time: float = Query(
        default=0.5,
        ge=0.05,
        le=10.0,
        description="Thời gian xử lý 1 công việc (giây)",
    ),
    max_queue_limit: int = Query(
        default=100,
        ge=1,
        le=1000,
        description="Giới hạn hàng đợi tối đa trước khi báo lỗi quá tải (503)",
    ),
):
    """
    Endpoint 2: Mô phỏng công việc xử lý theo hàng đợi (Queue).
    Chiếm rất ít CPU nhưng gây nghẽn hàng đợi và tăng độ trễ khi có nhiều request.
    Dùng để chứng minh HPA (theo CPU) thất bại và KEDA (theo Queue/In-flight) hiệu quả.
    """
    # Nếu hàng đợi đã đầy vượt mức cho phép, lập tức trả về lỗi 503 (Service Unavailable)
    # Phục vụ đo tỷ lệ lỗi (Error Rate) và quay demo tình huống lỗi
    current_waiting = QUEUE_LENGTH._value.get()
    if current_waiting >= max_queue_limit:
        PROCESSED_JOBS_TOTAL.labels(endpoint="queue", status="rejected_overload").inc()
        raise HTTPException(
            status_code=503,
            detail=f"Hàng đợi đã quá tải ({int(current_waiting)}/{max_queue_limit}). Vui lòng thử lại sau.",
        )

    # Yêu cầu bắt đầu xếp hàng chờ đến lượt
    QUEUE_LENGTH.inc()
    is_waiting_in_queue = True 
    enqueue_time = time.perf_counter()

    try:
        async with worker_semaphore:
            # Khi đã vào được đây nghĩa là đã tới lượt xử lý -> Rời khỏi hàng đợi chờ
            QUEUE_LENGTH.dec()
            is_waiting_in_queue = False

            wait_time = time.perf_counter() - enqueue_time
            QUEUE_WAIT_DURATION.observe(wait_time)

            # Mô phỏng thời gian xử lý công việc
            await asyncio.sleep(work_time)

            total_time = time.perf_counter() - enqueue_time
            PROCESSED_JOBS_TOTAL.labels(endpoint="queue", status="success").inc()

            return {
                "endpoint": "queue",
                "queue_wait_seconds": round(wait_time, 4),
                "work_time_seconds": work_time,
                "total_response_seconds": round(total_time, 4),
            }
    except asyncio.CancelledError:
        # Ghi nhận trường hợp client chủ động ngắt kết nối (timeout) khi đang chờ
        PROCESSED_JOBS_TOTAL.labels(endpoint="queue", status="client_cancelled").inc()
        raise
    finally:
        # Nếu yêu cầu bị ngắt khi vẫn đang đứng trong hàng đợi, giảm bộ đếm đi 1
        if is_waiting_in_queue:
            QUEUE_LENGTH.dec()

# scripts/collect-metrics.py
"""Export aligned Prometheus/Kubernetes time series for one k6 experiment."""
from __future__ import annotations

import argparse
import csv
import json
import math
import sys
import urllib.error
import urllib.parse
import urllib.request
from datetime import datetime, timezone
from pathlib import Path

QUERIES = {
    "replicas_desired": 'kube_deployment_spec_replicas{namespace="default",deployment="web-service"}',
    "replicas_available": 'kube_deployment_status_replicas_available{namespace="default",deployment="web-service"}',
    "cpu_cores": 'sum(rate(container_cpu_usage_seconds_total{namespace="default",pod=~"web-service-.*",container="web-service"}[1m]))',
    "memory_bytes": 'sum(container_memory_working_set_bytes{namespace="default",pod=~"web-service-.*",container="web-service"})',
    "queue_length": 'sum(app_queue_length{namespace="default",service="web-service"})',
    "in_flight": 'sum(app_in_flight_requests{namespace="default",service="web-service"})',
    "app_rps": 'sum(rate(http_requests_total{namespace="default",service="web-service",handler=~"/api/.*"}[1m]))',
}


def request_json(url: str, timeout: float) -> dict:
    with urllib.request.urlopen(url, timeout=timeout) as response:
        return json.load(response)


def fetch_range(base: str, query: str, start: float, end: float, step: int, timeout: float):
    args = urllib.parse.urlencode({"query": query, "start": start, "end": end, "step": step})
    result = request_json(base.rstrip('/') + '/api/v1/query_range?' + args, timeout)
    if result.get('status') != 'success':
        raise RuntimeError(str(result.get('error', result)))
    return result.get('data', {}).get('result', [])


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--prometheus-url', default='http://127.0.0.1:9090')
    parser.add_argument('--start', type=float, required=True, help='UTC Unix seconds')
    parser.add_argument('--end', type=float, required=True, help='UTC Unix seconds')
    parser.add_argument('--step', type=int, default=5)
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--timeout', type=float, default=20)
    args = parser.parse_args()
    if not (0 < args.start < args.end and args.step > 0):
        parser.error('require 0 < start < end and step > 0')
    args.output.mkdir(parents=True, exist_ok=True)
    rows = {}
    coverage = {}
    for metric, promql in QUERIES.items():
        try:
            series = fetch_range(args.prometheus_url, promql, args.start, args.end, args.step, args.timeout)
            coverage[metric] = {"series": len(series), "available": bool(series)}
            for item in series:
                for stamp, raw in item.get('values', []):
                    key = round(float(stamp), 3)
                    try:
                        value = float(raw)
                    except (ValueError, TypeError):
                        continue
                    if not math.isfinite(value):
                        continue
                    row = rows.setdefault(key, {"timestamp": key})
                    row[metric] = row.get(metric, 0) + value
        except (urllib.error.URLError, TimeoutError, ValueError, RuntimeError) as exc:
            coverage[metric] = {"series": 0, "available": False, "error": str(exc)}
            print(f'WARN {metric}: {exc}', file=sys.stderr)
    fieldnames = ['timestamp', 'timestamp_utc', *QUERIES]
    target = args.output / 'prometheus_timeseries.csv'
    with target.open('w', newline='', encoding='utf-8') as stream:
        writer = csv.DictWriter(stream, fieldnames=fieldnames)
        writer.writeheader()
        for stamp in sorted(rows):
            row = rows[stamp]
            row['timestamp_utc'] = datetime.fromtimestamp(stamp, timezone.utc).isoformat()
            writer.writerow(row)
    (args.output / 'prometheus_coverage.json').write_text(
        json.dumps({"queries": QUERIES, "coverage": coverage, "start": args.start,
                    "end": args.end, "step_seconds": args.step}, indent=2), encoding='utf-8')
    print(f'Wrote {len(rows)} timestamps to {target}')
    if not rows:
        print('ERROR: no numeric samples. Verify Prometheus URL, permissions and scrape.', file=sys.stderr)
        return 2
    if not coverage['replicas_available']['available']:
        print('WARN: kube-state-metrics may be disabled: replicas unavailable.', file=sys.stderr)
    if not coverage['cpu_cores']['available']:
        print('WARN: kubelet/cAdvisor may be disabled: CPU usage unavailable.', file=sys.stderr)
    return 0


if __name__ == '__main__':
    raise SystemExit(main())

function Invoke-Checked {
    param(
        [string]$Command,
        [string[]]$Arguments

    }

    # --------------------------------------------------------
    # 1. Validate required development tools
    # --------------------------------------------------------

    Write-Log INFO "Checking required development tools..."

    foreach ($tool in @("docker", "minikube", "kubectl", "helm")) {

            "start",
            "--profile=$Profile",
            "--driver=docker"
        )
    }
    else {

            "--driver=docker",
            "--cpus=$Cpus",
            "--memory=$MemoryMiB"

    # 4. Verify Kubernetes node readiness
    # --------------------------------------------------------

    Write-Log INFO "Checking Kubernetes node readiness..."

    Invoke-Checked -Command "kubectl" -Arguments @(

        "--context", $Profile,
        "-n", "kube-system",
        "rollout", "status",

        "--timeout=180s"
    )

    Write-Log OK "Metrics Server deployment is ready"

    # 6. Verify Kubernetes resource metrics
    # --------------------------------------------------------

    Write-Log INFO "Verifying resource metrics..."

    $metricsReady = $false
    $maxAttempts = 12

    for ($attempt = 1; $attempt -le $maxAttempts; $attempt++) {
        & kubectl --context $Profile top nodes

        if ($LASTEXITCODE -eq 0) {

            break
        }

        Write-Log WARN "Metrics API unavailable ($attempt/$maxAttempts)"

        }
    }

    # --------------------------------------------------------

    Write-Host ""

    Write-Host ""
    Write-Host "Cluster       : $Profile"
    Write-Host "Driver        : Docker"
    Write-Host "Node status   : Ready"
    Write-Host "Metrics Server: Available"
    Write-Host ""
}
catch {
    Write-Host ""
    Write-Log ERROR $_.Exception.Message
    exit 1
}

// load-tests/common.js
import http from 'k6/http';
import { check } from 'k6';
import { Counter, Rate, Trend } from 'k6/metrics';

export const applicationErrors = new Rate('app_error_rate');
export const overloadResponses = new Counter('overload_503_total');
export const queueWait = new Trend('queue_wait_seconds', true);
export const appProcessing = new Trend('app_processing_seconds', true);

function numberEnv(name, fallback, min, max) {
  const value = __ENV[name] === undefined ? fallback : Number(__ENV[name]);
  if (!Number.isFinite(value) || value < min || value > max) {
    throw new Error(`${name} must be between ${min} and ${max}`);
  }
  return value;
}

export const config = Object.freeze({
  baseUrl: (__ENV.BASE_URL || 'http://127.0.0.1:8000').replace(/\/+$/, ''),
  endpoint: (__ENV.ENDPOINT || 'queue').toLowerCase(),
  iterations: Math.round(numberEnv('ITERATIONS', 1000000, 1000, 20000000)),
  workTime: numberEnv('WORK_TIME', 0.5, 0.05, 10),
  queueLimit: Math.round(numberEnv('MAX_QUEUE_LIMIT', 100, 1, 1000)),
  timeout: numberEnv('REQUEST_TIMEOUT_SECONDS', 30, 1, 300),
  rate: numberEnv('RATE', 10, 1, 10000),
  maxVus: Math.round(numberEnv('MAX_VUS', 150, 1, 5000)),
});
if (!['cpu', 'queue'].includes(config.endpoint)) throw new Error('ENDPOINT must be cpu or queue');

export function requestWorkload() {
  const path = config.endpoint === 'cpu'
    ? `/api/cpu?iterations=${config.iterations}`
    : `/api/queue?work_time=${config.workTime}&max_queue_limit=${config.queueLimit}`;
  const res = http.get(`${config.baseUrl}${path}`, {
    timeout: `${config.timeout}s`,
    tags: { workload_endpoint: config.endpoint },
  });
  const ok = res.status === 200;
  applicationErrors.add(!ok);
  if (res.status === 503) overloadResponses.add(1);
  check(res, {
    'HTTP 200': (r) => r.status === 200,
    'response body agrees with endpoint': (r) => {
      if (r.status !== 200) return false;
      try { return r.json('endpoint') === config.endpoint; } catch (_) { return false; }
    },
  });
  if (ok) {
    try {
      const body = res.json();
      if (config.endpoint === 'queue' && Number.isFinite(body.queue_wait_seconds))
        queueWait.add(body.queue_wait_seconds);
      if (config.endpoint === 'cpu' && Number.isFinite(body.processing_time_seconds))
        appProcessing.add(body.processing_time_seconds);
    } catch (_) { /* HTTP validation above already detects invalid JSON */ }
  }
}

export function workloadThresholds(fault = false) {
  return {
    'http_req_duration': ['p(95)<30000'],
    'http_req_failed': [fault ? 'rate<1' : 'rate<0.8'],
    'dropped_iterations': ['count<1000000000'],
  };
}

export function saveSummary(data) {
  const filename = __ENV.SUMMARY_FILE || 'summary.json';
  return { [filename]: JSON.stringify(data, null, 2) };
}
