import asyncio
import math
import os
import time
from fastapi import FastAPI, HTTPException, Query, Request
from prometheus_client import Counter, Gauge, Histogram
from prometheus_fastapi_instrumentator import Instrumentator

# KHỞI TẠO ỨNG DỤNG & CẤU HÌNH CON ĐO (METRICS)
app = FastAPI(
    title="Autoscaling Web Service",
    description="Dịch vụ web stateless phục vụ thực nghiệm tự động co giãn (HPA & KEDA)",
    version="1.0.0",
)

# Con đo 1: Đếm số yêu cầu đang được xử lý cùng lúc
IN_FLIGHT_REQUESTS = Gauge(
    "app_in_flight_requests",
    "Số lượng yêu cầu đang được xử lý đồng thời tại thời điểm hiện tại",
)

# Con đo 2: Đếm số yêu cầu đang phải xếp hàng chờ
QUEUE_LENGTH = Gauge(
    "app_queue_length",
    "Số lượng công việc đang nằm chờ trong hàng đợi trước khi được xử lý",
)

# Con đo 3: Đếm tổng số công việc đã hoàn thành hoặc bị lỗi
PROCESSED_JOBS_TOTAL = Counter(
    "app_processed_jobs_total",
    "Tổng số yêu cầu đã đi qua hệ thống",
    ["endpoint", "status"],
)

# Con đo 4: Đo phân phối thời gian chờ trong hàng đợi
QUEUE_WAIT_DURATION = Histogram(
    "app_queue_wait_duration_seconds",
    "Thời gian một yêu cầu phải nằm chờ trong hàng đợi (tính bằng giây)",
    buckets=[0.01, 0.05, 0.1, 0.25, 0.5, 1.0, 2.0, 5.0, 10.0],
)

# Tự động gắn bộ thu thập chuẩn (độ trễ p95, throughput, error rate)
Instrumentator().instrument(app).expose(app, endpoint="/metrics")

# Đọc số lượng worker đồng thời từ biến môi trường
# Giúp dễ dàng thay đổi tham số thực nghiệm từ file deployment.yaml của Kubernetes
MAX_CONCURRENT_WORKERS = int(os.getenv("MAX_CONCURRENT_WORKERS", "5"))
worker_semaphore = asyncio.Semaphore(MAX_CONCURRENT_WORKERS)


# BỘ ĐẾM TỰ ĐỘNG CHO MỌI YÊU CẦU ĐI VÀO/ĐI RA
@app.middleware("http")
async def track_in_flight_requests(request: Request, call_next):
    # Bỏ qua không đếm các yêu cầu kiểm tra sức khỏe hoặc cào số liệu của hệ thống
    if request.url.path in ["/metrics", "/health"]:
        return await call_next(request)

    IN_FLIGHT_REQUESTS.inc()
    try:
        response = await call_next(request)
        return response
    finally:
        IN_FLIGHT_REQUESTS.dec()

# CÁC ĐƯỜNG DẪN API (ENDPOINTS) PHỤC VỤ THỰC NGHIỆM
@app.get("/health", tags=["System"])
async def health_check():
    """Kiểm tra ứng dụng còn sống hay không (phục vụ Kubernetes Liveness/Readiness Probe)."""
    return {"status": "healthy"}

@app.get("/api/cpu", tags=["Workloads"])
def simulate_cpu_load(
    iterations: int = Query(
        default=500_000,
        ge=1_000,
        le=20_000_000,
        description="Số vòng lặp tính toán để ép xung CPU",
    )
):
    """
    Endpoint 1: Tạo tải nặng về CPU.
    Dùng để kiểm thử chính sách co giãn theo CPU (Kubernetes HPA).
    """
    start_time = time.perf_counter()

    # Thực hiện phép tính căn bậc hai và lượng giác liên tục để tiêu tốn CPU thật
    accumulator = 0.0
    for i in range(1, iterations + 1):
        accumulator += math.sqrt(i) * math.sin(i)

    duration = time.perf_counter() - start_time
    PROCESSED_JOBS_TOTAL.labels(endpoint="cpu", status="success").inc()

    return {
        "endpoint": "cpu",
        "iterations": iterations,
        "processing_time_seconds": round(duration, 4),
    }

@app.get("/api/queue", tags=["Workloads"])
async def simulate_queue_load(
    work_time: float = Query(
        default=0.5,
        ge=0.05,
        le=10.0,
        description="Thời gian xử lý 1 công việc (giây)",
    ),
    max_queue_limit: int = Query(
        default=100,
        ge=1,
        le=1000,
        description="Giới hạn hàng đợi tối đa trước khi báo lỗi quá tải (503)",
    ),
):
    """
    Endpoint 2: Mô phỏng công việc xử lý theo hàng đợi (Queue).
    Chiếm rất ít CPU nhưng gây nghẽn hàng đợi và tăng độ trễ khi có nhiều request.
    Dùng để chứng minh HPA (theo CPU) thất bại và KEDA (theo Queue/In-flight) hiệu quả.
    """
    # Nếu hàng đợi đã đầy vượt mức cho phép, lập tức trả về lỗi 503 (Service Unavailable)
    # Phục vụ đo tỷ lệ lỗi (Error Rate) và quay demo tình huống lỗi
    current_waiting = QUEUE_LENGTH._value.get()
    if current_waiting >= max_queue_limit:
        PROCESSED_JOBS_TOTAL.labels(endpoint="queue", status="rejected_overload").inc()
        raise HTTPException(
            status_code=503,
            detail=f"Hàng đợi đã quá tải ({int(current_waiting)}/{max_queue_limit}). Vui lòng thử lại sau.",
        )

    # Yêu cầu bắt đầu xếp hàng chờ đến lượt
    QUEUE_LENGTH.inc()
    is_waiting_in_queue = True 
    enqueue_time = time.perf_counter()

    try:
        async with worker_semaphore:
            # Khi đã vào được đây nghĩa là đã tới lượt xử lý -> Rời khỏi hàng đợi chờ
            QUEUE_LENGTH.dec()
            is_waiting_in_queue = False

            wait_time = time.perf_counter() - enqueue_time
            QUEUE_WAIT_DURATION.observe(wait_time)

            # Mô phỏng thời gian xử lý công việc
            await asyncio.sleep(work_time)

            total_time = time.perf_counter() - enqueue_time
            PROCESSED_JOBS_TOTAL.labels(endpoint="queue", status="success").inc()

            return {
                "endpoint": "queue",
                "queue_wait_seconds": round(wait_time, 4),
                "work_time_seconds": work_time,
                "total_response_seconds": round(total_time, 4),
            }
    except asyncio.CancelledError:
        # Ghi nhận trường hợp client chủ động ngắt kết nối (timeout) khi đang chờ
        PROCESSED_JOBS_TOTAL.labels(endpoint="queue", status="client_cancelled").inc()
        raise
    finally:
        # Nếu yêu cầu bị ngắt khi vẫn đang đứng trong hàng đợi, giảm bộ đếm đi 1
        if is_waiting_in_queue:
            QUEUE_LENGTH.dec()

# ĐÁNH GIÁ CHI PHÍ TÀI NGUYÊN VÀ BẢO MẬT HỆ THỐNG

## 1. Tổng quan

Đề tài nghiên cứu tối ưu tự động co giãn dịch vụ web trên Kubernetes theo tải CPU và các chỉ số nghiệp vụ.

Hạ tầng thử nghiệm sử dụng Minikube trên máy cá nhân, kết hợp:

- Docker Desktop và Kubernetes.
- Kubernetes Metrics Server.
- Prometheus và Grafana.
- Prometheus Operator.
- KEDA Operator và External Metrics API.

Mục đích của tài liệu:

1. Xác định chi phí và giới hạn tài nguyên của môi trường thử nghiệm.
2. Phân tích các yếu tố ảnh hưởng đến khả năng mở rộng.
3. Đánh giá những rủi ro bảo mật khi triển khai.
4. Đề xuất biện pháp tối ưu và kiểm soát.
5. Xác định các điều kiện cần đáp ứng trước khi chuyển sang production.

## 2. Phạm vi đánh giá

Có hai phạm vi cần phân biệt.

### 2.1. Môi trường local

Sử dụng máy tính cá nhân và Minikube để chạy thử nghiệm.

Ưu điểm:

- Không cần thuê Kubernetes cluster trên cloud.
- Chủ động kiểm soát môi trường.
- Có thể cài đặt và dọn thử nghiệm nhiều lần.
- Phù hợp để xây dựng và kiểm thử các cấu hình autoscaling.

Hạn chế:

- Phụ thuộc phần cứng máy cá nhân.
- Một node không có tính sẵn sàng cao.
- CPU, RAM và I/O của các thành phần dùng chung tài nguyên.
- Khả năng mở rộng số Pod bị giới hạn bởi node.
- Kết quả hiệu năng không đại diện hoàn toàn cho production.

### 2.2. Môi trường production

Khi chuyển sang môi trường cloud hoặc Kubernetes nhiều node, cần bổ sung:

- High availability.
- Persistent storage.
- Network security.
- Quản lý secrets và phân quyền.
- Monitoring, alerting và logging đầy đủ.
- Backup và disaster recovery.
- Quản lý chi phí thực tế.
- Kiểm thử hiệu năng và bảo mật trước khi đưa vào vận hành.

Cấu hình local hiện tại không được mặc định xem là cấu hình production-ready.

## 5. Tối ưu footprint của Monitoring Stack

### 5.1. Giảm số lượng thành phần

Môi trường local đã tắt:

- Alertmanager.
- Default Prometheus rules.
- Default Grafana dashboards.
- kube-state-metrics.
- node-exporter.
- Một số thành phần thu thập metrics hệ thống.

Lợi ích là giảm số Pod, số tiến trình và overhead theo dõi.

Đánh đổi là hệ thống không có đầy đủ các dashboard, alert và chỉ số hạ tầng mặc định.

Các thành phần này cần được đánh giá lại khi chuyển sang môi trường production.

### 5.2. Giới hạn retention

Prometheus sử dụng retention 3 giờ để giảm nhu cầu lưu trữ lịch sử dài hạn.

Giới hạn:

- Không phù hợp để phân tích xu hướng dài ngày.
- Dữ liệu thử nghiệm cũ có thể bị loại bỏ.
- Không thay thế hệ thống lưu trữ kết quả nghiên cứu.

Với các đợt load test, dữ liệu kết quả cần được xuất và lưu riêng trước khi hết thời gian retention.

### 5.3. Không sử dụng persistence cho Grafana

Việc tắt persistence giúp đơn giản hóa môi trường local.

Đổi lại, các thay đổi được lưu trực tiếp trong Grafana có thể mất khi Pod bị tạo lại.

Dashboard và cấu hình quan trọng nên được quản lý bằng file và version control, thay vì chỉ lưu trong container.

## 6. Phân tích chi phí

### 6.1. Chi phí sử dụng Kubernetes local

Môi trường Minikube không phát sinh khoản phí thuê Kubernetes cluster từ nhà cung cấp cloud chỉ vì chạy local.

Tuy nhiên vẫn có chi phí:

- Điện năng tiêu thụ.
- Hao mòn và sử dụng phần cứng.
- Dung lượng lưu trữ.
- Internet để tải dependencies và container images.
- Thời gian cài đặt, vận hành và khắc phục sự cố.

Vì vậy không nên kết luận tổng chi phí vận hành bằng 0.

### 6.2. Ước tính chi phí điện năng

Công thức:

```text
Energy_kWh = Power_kW × Runtime_hours

ElectricityCost = Energy_kWh × ElectricityPrice_per_kWh
```

Nếu chỉ đánh giá điện năng tăng thêm do thí nghiệm:

```text
IncrementalEnergy_kWh =
    (PowerDuringExperiment_W - BaselinePower_W)
    × Runtime_hours
    / 1000
```

Không sử dụng số liệu công suất giả định làm kết quả đo thực tế.

Cần đo công suất hoặc sử dụng dữ liệu tiêu thụ có thể kiểm chứng.

### 6.3. Ước tính chi phí trên cloud

Khi chuyển sang cloud, các nhóm chi phí chính gồm:

| Nhóm | Yếu tố ảnh hưởng |
|---|---|
| Compute | Số node, loại máy, thời gian chạy |
| Storage | Dung lượng, loại ổ, thời gian lưu |
| Network | Data transfer và load balancer |
| Monitoring | Lưu trữ metrics/logs và dịch vụ quan sát |
| Control plane | Chính sách tính phí của nhà cung cấp |
| Backup | Dung lượng và tần suất sao lưu |
| Operations | Nhân lực và quản trị hệ thống |

Công thức tổng quát:

```text
TotalCost =
    ComputeCost
    + StorageCost
    + NetworkCost
    + MonitoringCost
    + ControlPlaneCost
    + BackupCost
    + OtherOperationalCost
```

Đơn giá phải lấy theo khu vực, cấu hình và nhà cung cấp tại thời điểm đánh giá.

Tài liệu này chưa đưa ra tổng chi phí cloud cụ thể vì nhóm chưa chốt cấu hình triển khai và bảng giá tương ứng.

## 7. Quan hệ giữa autoscaling và chi phí

Autoscaling có thể cải thiện hiệu quả sử dụng tài nguyên bằng cách:

- Tăng số replica khi tải vượt ngưỡng.
- Giảm replica khi nhu cầu giảm.
- Hạn chế cấp phát tài nguyên dư thừa trong thời gian tải thấp.
- Duy trì khả năng phục vụ trong các giai đoạn tải cao.

Tuy nhiên autoscaling không tự động đồng nghĩa với giảm chi phí.

Ví dụ, một cấu hình scale quá nhạy có thể tạo thêm Pod thường xuyên hoặc duy trì nhiều replica hơn cần thiết.

Khi có kết quả thực nghiệm, cần đánh giá đồng thời:

- Số replica trung bình.
- Số replica tối đa.
- Thời gian duy trì từng mức replica.
- CPU và memory usage.
- Request throughput.
- Latency p95.
- Error rate.
- Thời gian phản ứng với biến động tải.

## 8. Phương pháp so sánh chi phí tài nguyên

Dự kiến so sánh ba cấu hình:

| Cấu hình | Mục tiêu |
|---|---|
| CPU HPA | Baseline autoscaling theo CPU |
| KEDA chưa tối ưu | Autoscaling theo custom metrics với cấu hình ban đầu |
| KEDA tối ưu | Cấu hình custom metrics đã điều chỉnh ngưỡng và hành vi scale |

Các cấu hình cần được kiểm thử trên điều kiện tương đương:

- Cùng ứng dụng và container image.
- Cùng tài nguyên node.
- Cùng giới hạn tài nguyên Pod.
- Cùng kịch bản tải.
- Cùng thời lượng và phương pháp đo.
- Có giai đoạn warm-up và reset phù hợp.

Chỉ số chi phí tài nguyên có thể tính:

```text
ReplicaSeconds =
    Sum(ReplicaCount × DurationSeconds)
```

Hoặc đối với dữ liệu được lấy mẫu theo thời gian:

```text
ReplicaSeconds ≈
    Sum(ReplicaCount_i × SampleIntervalSeconds)
```

Chỉ số này biểu diễn mức sử dụng replica theo thời gian. Nó không trực tiếp bằng chi phí cloud nếu mỗi replica có resource request khác nhau hoặc dùng chung node với các workload khác.

Kết quả định lượng sẽ được bổ sung sau giai đoạn load testing.

## 9. Đánh giá bảo mật

### 9.1. Secret management

Grafana lưu thông tin đăng nhập quản trị trong Kubernetes Secret.

Rủi ro:

- Lộ mật khẩu khi sao chép output terminal.
- Commit nhầm Secret hoặc credentials lên GitHub.
- Phân quyền đọc Secret quá rộng.
- Sử dụng lại mật khẩu trong môi trường khác.

Biện pháp:

- Không hardcode credentials vào repository.
- Không lưu mật khẩu trong ảnh báo cáo.
- Chỉ cấp quyền đọc Secret cho đối tượng cần thiết.
- Sử dụng cơ chế quản lý Secret chuyên dụng trong production.
- Bật mã hóa Secret at rest khi môi trường hỗ trợ.

Kubernetes Secret được mã hóa Base64 khi biểu diễn trong YAML; Base64 không phải mã hóa bảo mật.

### 9.2. Network exposure

Các thành phần monitoring local sử dụng Kubernetes ClusterIP Service.

Khi cần truy cập từ máy host, sử dụng kubectl port-forward đến localhost.

Rủi ro:

- Vô tình công khai giao diện quản trị.
- Mở cổng không có xác thực phù hợp.
- Truy cập trái phép vào metrics nội bộ.

Biện pháp:

- Không tạo public LoadBalancer/Ingress khi chưa cần.
- Giới hạn port-forward ở localhost.
- Bảo vệ tài khoản Grafana.
- Áp dụng NetworkPolicy và TLS phù hợp trên production.

### 9.3. Prometheus Operator admission webhooks

Cấu hình local hiện tại:

```yaml
prometheusOperator:
  admissionWebhooks:
    enabled: false
  tls:
    enabled: false
```

Cấu hình này đã được sử dụng để khắc phục lỗi thiếu TLS Secret khi triển khai thử nghiệm.

Hạn chế:

- Không sử dụng cơ chế admission webhook của Prometheus Operator để kiểm tra các PrometheusRule theo cách cấu hình mặc định.
- Không đáp ứng đầy đủ yêu cầu bảo mật và kiểm soát cấu hình của production.

Định hướng:

- Bật lại admission webhooks khi môi trường hỗ trợ đầy đủ.
- Cấu hình TLS/certificates đúng cách.
- Kiểm thử quá trình cài đặt, nâng cấp và xác minh các rule.
- Không sao chép nguyên cấu hình local sang production mà không đánh giá lại.

### 9.4. KEDA Operator

Môi trường đã triển khai:

- KEDA Operator.
- Metrics API Server.
- Admission Webhooks.
- Các Kubernetes Custom Resource Definitions của KEDA.

External Metrics API đã được kiểm tra ở trạng thái Available.

Các biện pháp cần bổ sung khi tích hợp scaler:

- Phân quyền phù hợp cho service accounts.
- Hạn chế quyền truy cập metrics và secrets.
- Kiểm tra độ tin cậy của dữ liệu dùng để autoscaling.
- Giới hạn maxReplicaCount.
- Cấu hình authentication cho nguồn metrics khi cần.
- Kiểm thử tình huống scaler không truy cập được metric.

Không đặt cả HPA độc lập và KEDA ScaledObject cùng điều khiển số replica của một Deployment trong cùng phép thử nếu chưa thiết kế rõ cơ chế phối hợp.

### 9.5. Supply chain security

Rủi ro:

- Container images chứa lỗ hổng.
- Sử dụng phiên bản chart không kiểm soát.
- Tải dependencies từ nguồn không đáng tin cậy.
- Commit file cấu hình có thông tin nhạy cảm.

Biện pháp:

- Cố định phiên bản Helm charts.
- Ưu tiên image từ nguồn tin cậy.
- Quét dependencies và container images.
- Kiểm soát thay đổi thông qua Git và pull request.
- Xem xét image digest và chính sách xác minh artifact đối với production.

## 10. Rủi ro kỹ thuật và biện pháp kiểm soát

| Rủi ro | Tác động | Biện pháp |
|---|---|---|
| Node thiếu RAM | Pod Pending/OOMKilled | Resource requests/limits, theo dõi node |
| Grafana khởi động chậm | Probe thất bại | StartupProbe |
| Sai cấu hình TLS Operator | Pod không mount được Secret | Kiểm tra Helm values và manifest |
| Prometheus hết retention | Mất dữ liệu lịch sử | Xuất kết quả sau thí nghiệm |
| Grafana không persistence | Mất dashboard chỉnh tay | Quản lý dashboard bằng file |
| KEDA mất kết nối metrics | Autoscaling không như dự kiến | Giám sát scaler, kiểm thử lỗi |
| Scale quá nhạy | Tốn tài nguyên, dao động replica | Điều chỉnh threshold và cooldown |
| Metrics không đại diện tải thật | Quyết định scale sai | Kiểm thử metrics và PromQL |
| Một node bị lỗi | Dừng toàn bộ hệ thống | Chuyển multi-node trên production |
| Lộ thông tin nhạy cảm | Rủi ro bảo mật | Secrets, RBAC và kiểm soát Git |

## 11. Giới hạn của mô hình đo hiện tại

Môi trường là single-node Minikube, do đó:

1. Không thể đánh giá high availability giữa nhiều node.
2. Kết quả autoscaling bị ảnh hưởng bởi tài nguyên máy host.
3. Thử nghiệm chưa phản ánh đầy đủ network latency và chi phí cloud.
4. Monitoring Stack cùng chia sẻ tài nguyên với ứng dụng.
5. Retention ngắn không thích hợp để lưu kết quả dài hạn.
6. Chưa có bằng chứng đo chi phí cloud trực tiếp.

Đối với custom metrics, cần đặc biệt kiểm tra cách tổng hợp chỉ số từ nhiều Pod.

Nếu ứng dụng sử dụng bộ đếm hoặc hàng đợi mô phỏng trong bộ nhớ từng process, không được mặc định xem đó là hàng đợi phân tán hoặc dữ liệu tổng hợp toàn cluster.

## 12. Khuyến nghị trước khi triển khai production

- Sử dụng Kubernetes version nằm trong phạm vi tương thích đã kiểm thử của KEDA.
- Tách môi trường production khỏi local development.
- Bật lại các cơ chế bảo mật đã giản lược.
- Bổ sung TLS, RBAC, NetworkPolicy và secret management.
- Thiết lập persistent storage và backup.
- Bật alerting và monitoring phù hợp.
- Thiết lập resource quotas và limits.
- Sử dụng nhiều node nếu có yêu cầu high availability.
- Kiểm thử load, stress, soak và recovery.
- Đo chi phí thực tế và xây dựng ngưỡng ngân sách.
- Kiểm tra rollback và disaster recovery.

## 13. Kết luận

Môi trường Kubernetes local đã triển khai thành công Monitoring Stack và KEDA, đáp ứng yêu cầu hạ tầng ban đầu để nghiên cứu autoscaling.

Việc tinh giản Monitoring Stack, giới hạn tài nguyên và sử dụng Minikube giúp phù hợp với điều kiện phần cứng hiện có.

Tuy nhiên, thành công ở mức hạ tầng không đồng nghĩa với việc giải pháp autoscaling đã được chứng minh hiệu quả.

Để kết luận cấu hình tối ưu hơn, nhóm phải hoàn thành thí nghiệm, đo latency, error rate, replica usage và chi phí tài nguyên trên các kịch bản tải tương đương.

Bảo mật production cũng cần được đánh giá độc lập, không sử dụng nguyên trạng cấu hình local đã giản lược.

# HƯỚNG DẪN TÁI LẬP MÔI TRƯỜNG THỬ NGHIỆM

## 1. Mục đích

Tài liệu mô tả quy trình tái lập môi trường Kubernetes cục bộ phục vụ đề tài tối ưu tự động co giãn dịch vụ web theo tải và các chỉ số nghiệp vụ.

Mục tiêu:

- Khởi tạo Kubernetes cluster bằng Minikube.
- Bật Metrics Server để theo dõi tài nguyên CPU/RAM.
- Triển khai Prometheus, Grafana và Prometheus Operator.
- Triển khai KEDA Operator và External Metrics API.
- Kiểm tra trạng thái hoạt động của hạ tầng.
- Hỗ trợ dọn môi trường thử nghiệm có kiểm soát.

Phạm vi hiện tại là hạ tầng thử nghiệm. Việc triển khai ứng dụng web, cấu hình HPA, KEDA ScaledObject và kiểm thử autoscaling thực tế thuộc các giai đoạn tích hợp tiếp theo.

## 2. Kiến trúc môi trường

```text
Windows Host
    |
    +-- Docker Desktop (Hyper-V backend)
            |
            +-- Minikube Cluster
                    |
                    +-- Kubernetes Metrics Server
                    |
                    +-- Namespace: monitoring
                    |       |
                    |       +-- Prometheus
                    |       +-- Prometheus Operator
                    |       +-- Grafana
                    |
                    +-- Namespace: keda
                            |
                            +-- KEDA Operator
                            +-- KEDA Metrics API Server
                            +-- KEDA Admission Webhooks
```

Trong các giai đoạn sau, ứng dụng FastAPI sẽ được triển khai thành Kubernetes Deployment và Service. Prometheus sẽ thu thập các chỉ số ứng dụng khi cấu hình ServiceMonitor phù hợp. HPA và KEDA sẽ được kiểm thử theo từng kịch bản riêng.

## 3. Phiên bản và cấu hình tham chiếu

| Thành phần | Phiên bản/cấu hình tham chiếu |
|---|---|
| Hệ điều hành | Windows 10 Pro |
| Docker Desktop | 4.97.0 |
| Docker backend | Hyper-V |
| Giới hạn Docker | 4 GiB RAM, 4 CPU |
| Minikube | v1.39.0 |
| Kubernetes cluster đã kiểm thử | v1.37.0 |
| Minikube profile | minikube |
| Minikube driver | docker |
| Minikube CPU | 2 CPU |
| Minikube RAM | 3072 MiB |
| kubectl | v1.34.1 |
| Helm | v4.3.1 |
| kube-prometheus-stack chart | 92.16.1 |
| KEDA chart | 2.21.0 |

Đây là cấu hình tham chiếu đã sử dụng trong quá trình triển khai, không phải yêu cầu cố định cho mọi máy.

Lưu ý tương thích: KEDA 2.21 được kiểm thử chính thức với Kubernetes 1.34–1.36. Cluster Kubernetes 1.37 trong môi trường này đã chạy được các thành phần hạ tầng KEDA nhưng chưa được kiểm thử đầy đủ mọi chức năng autoscaling.

.\scripts\setup-cluster.ps1

Lưu ý: Nếu profile Minikube đã tồn tại, script có thể sử dụng lại cấu hình profile đó. Cần kiểm tra cấu hình thực tế thay vì mặc định rằng mọi lần chạy đều tạo cluster mới với cùng tài nguyên.

```powershell
kubectl --context minikube get nodes -o wide
kubectl --context minikube get pods -n kube-system
kubectl --context minikube top nodes
```

Kết quả mong muốn:

- Node `minikube` ở trạng thái `Ready`.
- Metrics Server hoạt động.
- `kubectl top nodes` trả về số liệu CPU và RAM.

## 12. Xử lý sự cố thường gặp

### 12.1. Helm timeout

Hiện tượng:

```text
context deadline exceeded
```

Nguyên nhân có thể do Pod chưa Ready, đang tải image, thiếu tài nguyên hoặc lỗi cấu hình.

Kiểm tra:

```powershell
kubectl --context minikube get pods -A
kubectl --context minikube get events -A --sort-by=.metadata.creationTimestamp
```

Không tự động xóa cluster hoặc cài lại Docker.

### 12.2. Prometheus Operator thiếu TLS Secret

Hiện tượng từng gặp:

```text
secret "monitoring-kube-prometheus-admission" not found
```

Cấu hình local đã giải quyết bằng cách tắt đồng thời:

```yaml
prometheusOperator:
  admissionWebhooks:
    enabled: false
  tls:
    enabled: false
```

Đây là lựa chọn dành cho môi trường thử nghiệm, không mặc định áp dụng cho production.

### 12.3. Grafana khởi động chậm

Hiện tượng:

```text
Liveness probe failed
connect: connection refused
```

Cấu hình local sử dụng startupProbe để dành thời gian cho Grafana hoàn tất khởi tạo trước khi Kubernetes áp dụng liveness probe.

Kiểm tra log và trạng thái Pod:

```powershell
kubectl --context minikube get pods -n monitoring
kubectl --context minikube describe deployment monitoring-grafana -n monitoring
```

Không kết luận lỗi thiếu RAM nếu chưa có bằng chứng OOMKilled hoặc sự kiện tài nguyên liên quan.

### 12.4. Pod ở trạng thái Pending

Kiểm tra:

```powershell
kubectl --context minikube describe pods -n monitoring
kubectl --context minikube top nodes
```

Đặc biệt chú ý thông báo Insufficient memory, Insufficient cpu, lỗi volume và lỗi tải image.

// load-tests/common.js
import http from 'k6/http';
import { check } from 'k6';
import { Counter, Rate, Trend } from 'k6/metrics';

export const applicationErrors = new Rate('app_error_rate');
export const overloadResponses = new Counter('overload_503_total');
export const queueWait = new Trend('queue_wait_seconds', true);
export const appProcessing = new Trend('app_processing_seconds', true);

function numberEnv(name, fallback, min, max) {
  const value = __ENV[name] === undefined ? fallback : Number(__ENV[name]);
  if (!Number.isFinite(value) || value < min || value > max) {
    throw new Error(`${name} must be between ${min} and ${max}`);
  }
  return value;
}

export const config = Object.freeze({
  baseUrl: (__ENV.BASE_URL || 'http://127.0.0.1:8000').replace(/\/+$/, ''),
  endpoint: (__ENV.ENDPOINT || 'queue').toLowerCase(),
  iterations: Math.round(numberEnv('ITERATIONS', 1000000, 1000, 20000000)),
  workTime: numberEnv('WORK_TIME', 0.5, 0.05, 10),
  queueLimit: Math.round(numberEnv('MAX_QUEUE_LIMIT', 100, 1, 1000)),
  timeout: numberEnv('REQUEST_TIMEOUT_SECONDS', 30, 1, 300),
  rate: numberEnv('RATE', 10, 1, 10000),
  maxVus: Math.round(numberEnv('MAX_VUS', 150, 1, 5000)),
});
if (!['cpu', 'queue'].includes(config.endpoint)) throw new Error('ENDPOINT must be cpu or queue');

export function requestWorkload() {
  const path = config.endpoint === 'cpu'
    ? `/api/cpu?iterations=${config.iterations}`
    : `/api/queue?work_time=${config.workTime}&max_queue_limit=${config.queueLimit}`;
  const res = http.get(`${config.baseUrl}${path}`, {
    timeout: `${config.timeout}s`,
    tags: { workload_endpoint: config.endpoint },
  });
  const ok = res.status === 200;
  applicationErrors.add(!ok);
  if (res.status === 503) overloadResponses.add(1);
  check(res, {
    'HTTP 200': (r) => r.status === 200,
    'response body agrees with endpoint': (r) => {
      if (r.status !== 200) return false;
      try { return r.json('endpoint') === config.endpoint; } catch (_) { return false; }
    },
  });
  if (ok) {
    try {
      const body = res.json();
      if (config.endpoint === 'queue' && Number.isFinite(body.queue_wait_seconds))
        queueWait.add(body.queue_wait_seconds);
      if (config.endpoint === 'cpu' && Number.isFinite(body.processing_time_seconds))
        appProcessing.add(body.processing_time_seconds);
    } catch (_) { /* HTTP validation above already detects invalid JSON */ }
  }
}

export function workloadThresholds(fault = false) {
  return {
    'http_req_duration': ['p(95)<30000'],
    'http_req_failed': [fault ? 'rate<1' : 'rate<0.8'],
    'dropped_iterations': ['count<1000000000'],
  };
}

export function saveSummary(data) {
  const filename = __ENV.SUMMARY_FILE || 'summary.json';
  return { [filename]: JSON.stringify(data, null, 2) };
}

# Đề tài 3: Tối ưu tự động co giãn cho dịch vụ web theo tải và theo chỉ số nghiệp vụ

Dự án nghiên cứu và thực nghiệm thuộc nhóm **Nền tảng và Kiến trúc Điện toán Đám mây**. Kho mã nguồn này lưu trữ toàn bộ mã nguồn dịch vụ, tệp cấu hình hạ tầng (Kubernetes/HPA/KEDA), kịch bản kiểm thử tải, dữ liệu đo thô và công cụ tái lập biểu đồ phân tích.

---

## 1. Tổng quan đề tài

### Câu hỏi nghiên cứu

> Chính sách co giãn theo CPU có đủ hiệu quả đối với tải biến động nhanh, hay cần kết hợp độ trễ, hàng đợi và số yêu cầu đang chờ?

### Mục tiêu nghiên cứu

- Thiết kế và so sánh ít nhất hai chính sách co giãn (theo CPU truyền thống và theo chỉ số tùy chỉnh/nghiệp vụ) cho cùng một dịch vụ web stateless.
- Đánh giá mối quan hệ đánh đổi giữa tốc độ phản ứng, chất lượng dịch vụ (QoS - p95 latency, error rate) và chi phí tài nguyên.
- Xác định ngưỡng (threshold) và khoảng ổn định (cooldown/stabilization window) nhằm giảm thiểu hiện tượng dao động (flapping/thrashing) tăng - giảm bản sao.

# Thành — load-test & data-analysis deliverable

This package adds files into the **existing** repository root; do not replace existing `app/`, `k8s/`, `monitoring/` or infrastructure scripts.

## Prerequisites
- Windows PowerShell, k6, Python 3.10+, Minikube/kubectl/Helm running.
- FastAPI deployed and reachable via `kubectl --context minikube -n default port-forward svc/web-service 8000:8000` in terminal 1.
- Prometheus reachable via `kubectl --context minikube -n monitoring port-forward svc/monitoring-kube-prometheus-prometheus 9090:9090` in terminal 2. Verify actual Service name with `kubectl -n monitoring get svc`.
- `python -m pip install -r requirements-analysis.txt`.

## Must check before each experiment
1. `kubectl --context minikube -n default get hpa,scaledobject,deploy,pods`.
2. **Only one policy** manages `web-service` Deployment. HPA CPU baseline: `kubectl --context minikube apply -f k8s/autoscaling/hpa-cpu.yaml` after deleting any KEDA ScaledObject. KEDA raw: delete standalone HPA and apply `k8s/autoscaling/keda-custom-raw.yaml`. KEDA optimized must be supplied by Huy; do not invent production results for an unprovided configuration.
3. Each run must begin with equivalent available replicas (ideally 1), same image, resource requests/limits, same k6 rate, same CPU/queue settings and wait for scale-down/cooldown between runs.
4. Verify http://127.0.0.1:8000/health and http://127.0.0.1:9090/api/v1/query?query=up. Run `kubectl get svc -n monitoring` to confirm Prometheus naming.
5. Monitoring chart may have disabled kube-state-metrics and kubelet/cAdvisor. If `prometheus_coverage.json` marks replica/CPU unavailable, enable relevant exporters/targets via the monitoring Helm values together with Huy before collecting final results. `kubectl top` alone does not prove those series exist in Prometheus.

## Execute
Run PowerShell from repository root:

```powershell
.\scripts\run-experiment.ps1 -Policy baseline-cpu -Scenario step -Endpoint queue -Repeat 1
.\scripts\run-experiment.ps1 -Policy custom-unoptimized -Scenario step -Endpoint queue -Repeat 1
# Only after Huy supplies and applies keda-optimized.yaml:
.\scripts\run-experiment.ps1 -Policy custom-optimized -Scenario step -Endpoint queue -Repeat 1
```

For all workloads run `step`, `spike`, `periodic` and both endpoints (`queue`, `cpu`) for each policy. Repeat >=3 times, ideally fresh cluster resource conditions. Use `-Rate 10 -MaxVus 150 -WorkTime 0.5 -Iterations 1000000` identically for comparable cases. For failure demo: `-Scenario fault -Endpoint queue` (forced queue work_time=1 and limit=10); do **not** merge fault results with ordinary benchmark summaries.

If you want raw k6 only: `$env:ENDPOINT='queue'; $env:BASE_URL='http://127.0.0.1:8000'; k6 run load-tests/step-workload.js`.

Raw results are stored per run under `data/<policy>/<run-id>/`: k6 summary, raw JSON, console log, metadata, Prometheus timeline and coverage. Real outputs are **not included** in this package because no live Minikube/Prometheus was supplied.

Open `notebooks/analysis.ipynb` in VS Code/Jupyter and Run All after obtaining real data. It produces `data/comparison-runs.csv`, `data/comparison-aggregated.csv` and figures in `notebooks/figures/`.

## Measurement caveats
- `http_req_duration` excludes some connection stages; use k6 metric definition consistently.
- k6 `app_error_rate` counts any non-200 response, including HTTP 503 and request failures. Collect HTTP 503 separately.
- `dropped_iterations` means offered load could not be executed and must be reported, not silently ignored.
- `replicas_available` requires kube-state-metrics; CPU and RAM require container metrics scraped by Prometheus.
- Replica timeline resolution equals Prometheus scrape step, cannot resolve changes finer than sampling interval.
- The application maintains an in-memory queue per Pod; KEDA is driven by summed per-Pod queue metric, not a shared external broker.

## Source ownership
This package does not modify files owned by Huy/Hiếu and it does not manufacture test results. Merge through branch `feat/loadtest-analysis`; coordinate autoscaler switching with Huy.

# Safely clean up Helm-managed monitoring components
# in the local Minikube development environment.
#
# Default: preview only.
# Destructive operations require -Execute and confirmation.

[CmdletBinding()]
param(
    [ValidateSet("All", "Keda", "Monitoring")]
    [string]$Scope = "All",

    [switch]$Execute
)

Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"

$KubeContext = "minikube"

function Assert-Command {
    param([string]$Name)

    if (-not (Get-Command $Name -ErrorAction SilentlyContinue)) {
        throw "Required command not found: $Name"
    }
}

function Get-ReleaseInstalled {
    param(
        [string]$Release,
        [string]$Namespace
    )

    $result = & helm list `
        --namespace $Namespace `
        --kube-context $KubeContext `
        --output json

    if ($LASTEXITCODE -ne 0) {
        throw "Failed to inspect Helm releases in namespace '$Namespace'."
    }

    $json = ($result | Out-String).Trim()

    if ([string]::IsNullOrWhiteSpace($json)) {
        return $false
    }

    $releases = @(ConvertFrom-Json -InputObject $json)

    foreach ($item in $releases) {
        if ($item.name -eq $Release) {
            return $true
        }
    }

    return $false
}

function Assert-NoKedaWorkloads {
    $resourceKinds = @(
        "scaledobjects.keda.sh",
        "scaledjobs.keda.sh"
    )

    foreach ($kind in $resourceKinds) {
        $result = & kubectl `
            --context $KubeContext `
            get $kind `
            --all-namespaces `
            --output json

        if ($LASTEXITCODE -ne 0) {
            throw "Unable to inspect '$kind'. KEDA cleanup aborted."
        }

        $data = ConvertFrom-Json -InputObject (
            ($result | Out-String).Trim()
        )

        if (@($data.items).Count -gt 0) {
            throw "Active '$kind' resources exist. KEDA cleanup aborted."
        }
    }
}

Write-Host "[INFO] Local Kubernetes cleanup"
Write-Host "[INFO] Context: $KubeContext"
Write-Host "[INFO] Scope: $Scope"

Assert-Command "helm"
Assert-Command "kubectl"

$contexts = @(& kubectl config get-contexts -o name)

if ($LASTEXITCODE -ne 0 -or $contexts -notcontains $KubeContext) {
    throw "Kubernetes context '$KubeContext' is unavailable."
}

$allTargets = @(
    [pscustomobject]@{
        Scope     = "Keda"
        Release   = "keda"
        Namespace = "keda"
    },
    [pscustomobject]@{
        Scope     = "Monitoring"
        Release   = "monitoring"
        Namespace = "monitoring"
    }
)

$selectedTargets = @(
    $allTargets | Where-Object {
        $Scope -eq "All" -or $_.Scope -eq $Scope
    }
)

$installedTargets = @()

foreach ($target in $selectedTargets) {
    $installed = Get-ReleaseInstalled `
        -Release $target.Release `
        -Namespace $target.Namespace

    if ($installed) {
        $installedTargets += $target

        Write-Host (
            "[FOUND] {0} in namespace {1}" -f `
            $target.Release, $target.Namespace
        )
    }
    else {
        Write-Host (
            "[SKIP] Release {0} is not installed." -f `
            $target.Release
        )
    }
}

if ($installedTargets.Count -eq 0) {
    Write-Host "[OK] Nothing to clean up."
    return
}

if (-not $Execute) {
    Write-Host ""
    Write-Host "[DRY RUN] No resources were deleted."
    Write-Host "[INFO] To uninstall the selected releases, use -Execute."
    return
}

# Protect existing autoscaling experiments.
if (@($installedTargets | Where-Object {
    $_.Release -eq "keda"
}).Count -gt 0) {
    Assert-NoKedaWorkloads
}

Write-Host ""
Write-Warning "The selected Helm releases will be uninstalled."
Write-Warning "Source code, Docker, Minikube and namespaces will be preserved."

$confirmation = Read-Host "Type UNINSTALL to continue"

if ($confirmation -cne "UNINSTALL") {
    Write-Host "[CANCELLED] No resources were deleted."
    return
}

foreach ($target in $installedTargets) {
    Write-Host (
        "[INFO] Uninstalling {0} from {1}..." -f `
        $target.Release, $target.Namespace
    )

    & helm uninstall $target.Release `
        --namespace $target.Namespace `
        --kube-context $KubeContext `
        --wait `
        --timeout 3m

    if ($LASTEXITCODE -ne 0) {
        throw "Helm uninstall failed for '$($target.Release)'."
    }

    Write-Host "[OK] Removed Helm release: $($target.Release)"
}

Write-Host ""
Write-Host "[OK] Cleanup completed."
Write-Host "[INFO] Minikube cluster and local source files were preserved."

# scripts/collect-metrics.py
"""Export aligned Prometheus/Kubernetes time series for one k6 experiment."""
from __future__ import annotations

import argparse
import csv
import json
import math
import sys
import urllib.error
import urllib.parse
import urllib.request
from datetime import datetime, timezone
from pathlib import Path

QUERIES = {
    "replicas_desired": 'kube_deployment_spec_replicas{namespace="default",deployment="web-service"}',
    "replicas_available": 'kube_deployment_status_replicas_available{namespace="default",deployment="web-service"}',
    "cpu_cores": 'sum(rate(container_cpu_usage_seconds_total{namespace="default",pod=~"web-service-.*",container="web-service"}[1m]))',
    "memory_bytes": 'sum(container_memory_working_set_bytes{namespace="default",pod=~"web-service-.*",container="web-service"})',
    "queue_length": 'sum(app_queue_length{namespace="default",service="web-service"})',
    "in_flight": 'sum(app_in_flight_requests{namespace="default",service="web-service"})',
    "app_rps": 'sum(rate(http_requests_total{namespace="default",service="web-service",handler=~"/api/.*"}[1m]))',
}


def request_json(url: str, timeout: float) -> dict:
    with urllib.request.urlopen(url, timeout=timeout) as response:
        return json.load(response)


def fetch_range(base: str, query: str, start: float, end: float, step: int, timeout: float):
    args = urllib.parse.urlencode({"query": query, "start": start, "end": end, "step": step})
    result = request_json(base.rstrip('/') + '/api/v1/query_range?' + args, timeout)
    if result.get('status') != 'success':
        raise RuntimeError(str(result.get('error', result)))
    return result.get('data', {}).get('result', [])


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--prometheus-url', default='http://127.0.0.1:9090')
    parser.add_argument('--start', type=float, required=True, help='UTC Unix seconds')
    parser.add_argument('--end', type=float, required=True, help='UTC Unix seconds')
    parser.add_argument('--step', type=int, default=5)
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--timeout', type=float, default=20)
    args = parser.parse_args()
    if not (0 < args.start < args.end and args.step > 0):
        parser.error('require 0 < start < end and step > 0')
    args.output.mkdir(parents=True, exist_ok=True)
    rows = {}
    coverage = {}
    for metric, promql in QUERIES.items():
        try:
            series = fetch_range(args.prometheus_url, promql, args.start, args.end, args.step, args.timeout)
            coverage[metric] = {"series": len(series), "available": bool(series)}
            for item in series:
                for stamp, raw in item.get('values', []):
                    key = round(float(stamp), 3)
                    try:
                        value = float(raw)
                    except (ValueError, TypeError):
                        continue
                    if not math.isfinite(value):
                        continue
                    row = rows.setdefault(key, {"timestamp": key})
                    row[metric] = row.get(metric, 0) + value
        except (urllib.error.URLError, TimeoutError, ValueError, RuntimeError) as exc:
            coverage[metric] = {"series": 0, "available": False, "error": str(exc)}
            print(f'WARN {metric}: {exc}', file=sys.stderr)
    fieldnames = ['timestamp', 'timestamp_utc', *QUERIES]
    target = args.output / 'prometheus_timeseries.csv'
    with target.open('w', newline='', encoding='utf-8') as stream:
        writer = csv.DictWriter(stream, fieldnames=fieldnames)
        writer.writeheader()
        for stamp in sorted(rows):
            row = rows[stamp]
            row['timestamp_utc'] = datetime.fromtimestamp(stamp, timezone.utc).isoformat()
            writer.writerow(row)
    (args.output / 'prometheus_coverage.json').write_text(
        json.dumps({"queries": QUERIES, "coverage": coverage, "start": args.start,
                    "end": args.end, "step_seconds": args.step}, indent=2), encoding='utf-8')
    print(f'Wrote {len(rows)} timestamps to {target}')
    if not rows:
        print('ERROR: no numeric samples. Verify Prometheus URL, permissions and scrape.', file=sys.stderr)
        return 2
    if not coverage['replicas_available']['available']:
        print('WARN: kube-state-metrics may be disabled: replicas unavailable.', file=sys.stderr)
    if not coverage['cpu_cores']['available']:
        print('WARN: kubelet/cAdvisor may be disabled: CPU usage unavailable.', file=sys.stderr)
    return 0


if __name__ == '__main__':
    raise SystemExit(main())

# scripts/run-experiment.ps1
[CmdletBinding()]

Write-Host "Run folder: $out" -ForegroundColor Cyan

    $end=[DateTimeOffset]::UtcNow.ToUnixTimeSeconds()
    $metadata.ended_unix=$end
    $metadata.k6_exit_code=$exitCode

    & $python.Source (Join-Path $root 'scripts/collect-metrics.py') --prometheus-url $PrometheusUrl --start ($begin - 30) --end ($end + 30) --step $SampleSeconds --output $out
    Write-Host "Metrics collector exit code: $LASTEXITCODE"
}
if ($exitCode -ne 0) { Write-Warning "k6 exit code $exitCode: check thresholds and logs; measurements were retained" }
Write-Host "Data: $out" -ForegroundColor Green

    }

    $dockerMemoryBytes = [long]::Parse(
        ($memoryOutput | Out-String).Trim()
    )

    $dockerMemoryMiB = [math]::Floor(
        $dockerMemoryBytes / 1MB
    )

    Write-Log OK "Docker Engine is running"
    Write-Log INFO "Docker memory limit: $dockerMemoryMiB MiB"

    # --------------------------------------------------------
    # 3. Create or reuse the Minikube cluster
    # --------------------------------------------------------

    Write-Log INFO "Checking Minikube profile: $Profile"

    $profilePath = Join-Path $env:USERPROFILE `
        ".minikube\profiles\$Profile\config.json"

    if (Test-Path $profilePath) {
        Write-Log INFO "Existing profile detected. Reusing cluster."

        Invoke-Checked -Command "minikube" -Arguments @(
            "start",
            "--profile=$Profile",
            "--driver=docker"
        )
    }
    else {
        $requiredMemoryMiB = $MemoryMiB + 512

        if ($dockerMemoryMiB -lt $requiredMemoryMiB) {
            throw "Insufficient Docker memory. Required: $requiredMemoryMiB MiB; available limit: $dockerMemoryMiB MiB."
        }

        Write-Log INFO "Creating Minikube cluster..."
        Write-Log INFO "CPU: $Cpus | Memory: $MemoryMiB MiB"

        Invoke-Checked -Command "minikube" -Arguments @(
            "start",
            "--profile=$Profile",
            "--driver=docker",
            "--cpus=$Cpus",
            "--memory=$MemoryMiB"
        )
    }

    # --------------------------------------------------------
    # 4. Verify Kubernetes node readiness
    # --------------------------------------------------------

    Write-Log INFO "Checking Kubernetes node readiness..."

    Invoke-Checked -Command "kubectl" -Arguments @(
        "--context", $Profile,
        "wait",
        "--for=condition=Ready",
        "nodes",
        "--all",
        "--timeout=180s"
    )

    Invoke-Checked -Command "kubectl" -Arguments @(
        "--context", $Profile,
        "get", "nodes", "-o", "wide"
    )

    Write-Log OK "Kubernetes nodes are ready"

    # --------------------------------------------------------
    # 5. Enable Kubernetes Metrics Server
    # --------------------------------------------------------

    Write-Log INFO "Enabling Metrics Server..."

    Invoke-Checked -Command "minikube" -Arguments @(
        "addons", "enable", "metrics-server",
        "--profile=$Profile"
    )

    Invoke-Checked -Command "kubectl" -Arguments @(
        "--context", $Profile,
        "-n", "kube-system",
        "rollout", "status",
        "deployment/metrics-server",
        "--timeout=180s"
    )

    Write-Log OK "Metrics Server deployment is ready"

    # --------------------------------------------------------
    # 6. Verify Kubernetes resource metrics
    # --------------------------------------------------------

    Write-Log INFO "Verifying resource metrics..."

    $metricsReady = $false
    $maxAttempts = 12

    for ($attempt = 1; $attempt -le $maxAttempts; $attempt++) {
        & kubectl --context $Profile top nodes

        if ($LASTEXITCODE -eq 0) {
            $metricsReady = $true
            break
        }

        Write-Log WARN "Metrics API unavailable ($attempt/$maxAttempts)"

        if ($attempt -lt $maxAttempts) {
            Start-Sleep -Seconds 5
        }
    }

    if (-not $metricsReady) {
        throw "Metrics API is unavailable. Inspect Metrics Server logs and APIService status."
    }

    # --------------------------------------------------------
    # 7. Display setup summary
    # --------------------------------------------------------

    Write-Host ""
    Write-Log OK "Kubernetes cluster setup completed successfully."
    Write-Host ""
    Write-Host "Cluster       : $Profile"
    Write-Host "Driver        : Docker"
    Write-Host "Node status   : Ready"
    Write-Host "Metrics Server: Available"
    Write-Host ""
}
catch {
    Write-Host ""
    Write-Log ERROR $_.Exception.Message
    exit 1
}
